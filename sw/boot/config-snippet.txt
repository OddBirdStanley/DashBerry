# Merge into /boot/firmware/config.txt (Pi 4, Raspberry Pi OS Lite 64-bit).
# cli/dashberry-install merges everything above the dashed separator
# automatically when it builds the card (--bypass-time drops the DS3231
# lines) — this snippet stays as the manual-install reference.

# --- front camera: CSI Camera Module 3 (auto-detect handles IMX708) ---
camera_auto_detect=1

# --- I2C at 400 kHz: 5 Hz OLED repaints need it; that's also the DS3231's
# fast-mode ceiling (shared bus: 0x3c panel + 0x68 RTC) ---
dtparam=i2c_arm=on,i2c_arm_baudrate=400000

# --- DS3231 RTC ---
dtoverlay=i2c-rtc,ds3231

# --- 128x64 OLED Bonnet (#3531): SSD1306 at 0x3C, Trixie binds the DRM
# ssd130x driver which exposes it as /dev/fbN — the number is probe order
# and cannot be assumed, so the panel resolves it by driver name via
# /sys/class/graphics/fb*/name. "inverted" is required (without it
# everything renders mirrored): it sets solomon,com-invdir, giving the
# COM-scan direction (0xC8) the bonnet is built for; the overlay's base
# node sets no orientation properties and the driver's segment-remap-on
# default (0xA1) covers the other axis. (Diagnostics note: /dev/i2c-*
# needs `modprobe i2c-dev` — nothing on the image loads it; i2cdetect
# then shows 0x3c/0x68 as UU, i.e. driver-claimed.) ---
dtoverlay=ssd1306,inverted

# --- no HDMI/audio needed headless; trims boot + power ---
dtoverlay=vc4-kms-v3d
disable_splash=1

# --- Bluetooth: OFF, on every card, with no exception. Nothing in
# DashBerry has ever used BT, and a dashcam whose README says it transmits
# nothing must not carry a radio no layer of the card asserts. Until this
# line, none did: every RF layer here is scoped to rfkill type wlan, so
# whether BT came up live was decided entirely by the base image.
#
# This is the layer that cannot be argued with — the firmware never
# attaches the BT modem, so there is no adapter for bluetoothd to power on
# or for a writable OS to un-block. Everything on the Linux side
# (dashberry-btkill.service, the udev rule, the masked bluetooth.service)
# is depth beneath it, and is what covers a card built before this line
# existed or one whose config.txt was hand-edited.
#
# Safe on this build, checked rather than assumed: disable-bt frees the
# PL011 UART and hands ttyAMA0 back to GPIO 14/15, and nothing on the card
# wants it — the GPS puck is USB (gpsd is pinned to the /dev/gps0 symlink
# over ttyACM), the panel is I2C, and the serial console is not enabled.
# It also leaves hciuart failing at boot with nothing to attach, which is
# why firstinstall masks that unit. ---
dtoverlay=disable-bt

# --- VideoCore firmware memory. The firmware-backed camera/codec contexts
# (bcm2835-isp for the front 1080p libcamera path and bcm2835-codec for its
# H.264 encode) allocate here, NOT from ARM-side CMA. The 76 MB
# default cannot hold them all: the last context to arrive fails —
# bench-observed both as a clean REQBUFS ENOMEM ("Failed to allocate
# required memory") and as a VCHIQ sync timeout that wedges the firmware
# until power-cycle, taking the ISP and front recording down with it.
# 256 MB fits everything (bench-verified). ---
#
# THE REAR IS NOT IN THIS BUDGET, and has not been since 2026-08-13. It used
# to be: the rear encoded H.264 through bcm2835-codec, which is a firmware
# context and did allocate here. Then the encode went to software openh264enc
# on this box, and then off this box entirely — the Zero encodes now and
# rear-rec is a remux, so there is no rear decode and no rear encode here to
# hold a context. Both times the rear left this budget it left it smaller,
# never larger, so 256 stays right by margin rather than by measurement. Do
# not read a rear camera into this figure if it needs re-deriving.
gpu_mem=256

# ---------------------------------------------------------------

#  - `fbcon=map:9` is appended by dashberry-install (the OLED enumerates
#    as fb0, fbcon's default target — boot tty text and a blinking
#    cursor land on the panel otherwise). fb9 never exists, so
#    fbcon binds nothing, on any probe order; the OLED's only writer is
#    the panel daemon.
#  - No consoleblank needed: the panel blanks itself (AUTO-BLANK).
